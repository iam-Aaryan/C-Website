#include <stdio.h>
int fact, rev, count, sum;
void factorial()
{
    int n, i;
    fact = 1;
    printf("\nEnter a number: ");
    scanf("%d", &n);
    for (i = 1; i <= n; i++)
        fact = fact * i;
    printf("Factorial = %d", fact);
}
void reverse()
{
    int n, rem;
    rev = 0;
    printf("\nEnter a number: ");
    scanf("%d", &n);
    while (n > 0)
    {
        rem = n % 10;
        rev = rev * 10 + rem;
        n = n / 10;
    }
    printf("Reverse = %d", rev);
}
void palindrome()
{
    int n, temp, rem;
    rev = 0;
    printf("\nEnter a number: ");
    scanf("%d", &n);
    temp = n;
    while (n > 0)
    {
        rem = n % 10;
        rev = rev * 10 + rem;
        n = n / 10;
    }
    if (temp == rev)
        printf("Palindrome");
    else
        printf("Not Palindrome");
}
void prime()
{
    int n, i;
    count = 0;
    printf("\nEnter a number: ");
    scanf("%d", &n);
    for (i = 2; i <= n / 2; i++)
    {
        if (n % i == 0)
        {
            count = 1;
            break;
        }
    }
    if (n > 1 && count == 0)
        printf("Prime Number");
    else
        printf("Not Prime Number");
}
void armstrong()
{
    int n, rem, temp;
    sum = 0;
    printf("\nEnter a number: ");
    scanf("%d", &n);
    temp = n;
    while (n > 0)
    {
        rem = n % 10;
        sum = sum + rem * rem * rem;
        n = n / 10;
    }
    if (temp == sum)
        printf("Armstrong Number");
    else
        printf("Not Armstrong Number");
}
void fibonacci()
{
    int a = 0, b = 1, c;
    printf("\nFibonacci Series:\n");
    while (a <= 100)
    {
        printf("%d ", a);
        c = a + b;
        a = b;
        b = c;
    }
}
void main()
{
    int ch;
    while (1)
    {
        printf("\n\n1. Factorial");
        printf("\n2. Reverse");
        printf("\n3. Palindrome");
        printf("\n4. Prime Number");
        printf("\n5. Armstrong");
        printf("\n6. Fibonacci");
        printf("\n7. Exit");

        printf("\nEnter your choice: ");
        scanf("%d", &ch);

        if (ch == 1)
            factorial();

        else if (ch == 2)
            reverse();

        else if (ch == 3)
            palindrome();

        else if (ch == 4)
            prime();

        else if (ch == 5)
            armstrong();

        else if (ch == 6)
            fibonacci();

        else if (ch == 7)
            break;

        else
            printf("Invalid Choice");
    }
}

#include <stdio.h>
void swap(int *a, int *b)
{
    int temp;

    temp = *a;
    *a = *b;
    *b = temp;
}
void main()
{
    int a, b;
    printf("Enter two numbers: ");
    scanf("%d%d", &a, &b);
    swap(&a, &b);
    printf("After swapping:\n");
    printf("a = %d\n", a);
    printf("b = %d", b);
}
Output :

Enter two numbers: 10 20

After swapping:
a = 20
b = 10

{
    int temp;
    temp = a;
    a = b;
    b = temp;

    printf("After swapping:\n");
    printf("a = %d\nb = %d", a, b);

#include <stdio.h>
void power(int num, int p)
{
    int result = 1;
    for(int i = 1; i <= p; i++)
    {
        result = result * num;
    }
    printf("Result = %d", result);
}
void main()
{
    int num, p;
    printf("Enter the number: ");
    scanf("%d", &num);
    printf("Enter the power: ");
    scanf("%d", &p);
    power(num, p);
}
Output :

Enter the number: 2
Enter the power: 3
Result = 8
