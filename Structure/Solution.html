#include<stdio.h>

struct employee
{
    int id;
    char name[50];
    char designation[50];
};

void main()
{
    struct employee e[50];
    int i;

    for(i=0; i<50; i++)
    {
        printf("\nEnter details of employee %d\n", i+1);

        printf("Enter ID: ");
        scanf("%d", &e[i].id);

        getchar();

        printf("Enter Name: ");
        gets(e[i].name);

        printf("Enter Designation: ");
        gets(e[i].designation);
    }

    printf("\nID\tName\t\tDesignation\n");

    for(i=0; i<50; i++)
    {
        printf("%d\t%s\t\t%s\n", e[i].id, e[i].name, e[i].designation);
    }
}

#include <stdio.h>

struct student
{
    int roll;
    char name[50];
    float fee;
    char dob[20];
};

void main()
{
    struct student s[50], temp;
    int i, j, n;

    printf("Enter number of students: ");
    scanf("%d", &n);

    for(i=0; i<n; i++)
    {
        printf("\nEnter details of student %d\n", i+1);

        printf("Enter roll no: ");
        scanf("%d", &s[i].roll);

        getchar();

        printf("Enter name: ");
        gets(s[i].name);

        printf("Enter fee: ");
        scanf("%f", &s[i].fee);

        getchar();

        printf("Enter DOB: ");
        gets(s[i].dob);
    }

    for(i=0; i < n-1; i++)
    {
        for(j=0; j<n-i-1; j++)
        {
            if(s[j].roll > s[j+1].roll)
            {
                temp = s[j];
                s[j] = s[j+1];
                s[j+1] = temp;
            }
        }
    }

    printf("\nRoll\tName\tFee\tDOB\n");

    for(i=0; i<n; i++)
    {
        printf("%d\t%s\t%.2f\t%s\n",
        s[i].roll, s[i].name, s[i].fee, s[i].dob);
    }
}
