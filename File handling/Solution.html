#include <stdio.h>
#include <conio.h>

void main()
{
    char n[20], d[100], res;
    char str[100], *c;
    int count = 0, i;
    FILE *fp;

    printf("Enter file name : ");
    gets(n);

    fp = fopen(n, "w");

    while (1)
    {
        printf("\nEnter the data : ");
        gets(d);
        fputs(d, fp);

        printf("Do you want to add more : ");
        res = getche();

        if (res == 'y' || res == 'Y')
        {
            fputs("\n", fp);
        }
        else
        {
            break;
        }
    }

    fclose(fp);

    fp = fopen(n, "r");

    while (1)
    {
        c = fgets(str, 100, fp);
        if (c != NULL)
        {
            for (i = 0; str[i] != '\0'; i++)
            {
                if (str[i] == ' ' || str[i] == '\n')
                {
                    count++;
                }
            }
            if (i > 0 && str[i - 1] != ' ' && str[i - 1] != '\n')
            {
                count++;
            }
        }
        else
        {
            break;
        }
    }
    printf("\n\nTotal number of words = %d", count);
    fclose(fp);
}
Output :

Enter file name : folder

Enter the data : Hello everyone
Do you want to add more : y

Enter the data : Welcome to C programming
Do you want to add more : y

Enter the data : File handling is interesting
Do you want to add more : n

Total number of words = 10
