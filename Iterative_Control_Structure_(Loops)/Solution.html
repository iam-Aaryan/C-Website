#include <stdio.h>
void main()
{
    int n, r, b = 0, p = 1;
    printf("Enter a number: ");
    scanf("%d", &n);

    {

}

{

            if (ch == '1')

            else if (ch == '2')
                printf("Subtraction = %d", a - b);

            else if (ch == '3')
                printf("Multiplication = %d", a * b);

            else if (ch == '4')
            {
                if (b != 0)
                    printf("Division = %d", a / b);
                else
                    printf("Cannot divide by zero");
            }

            else if (ch == '5')
            {
                if (b != 0)
                    printf("Modulo = %d", a % b);
                else
                    printf("Cannot divide by zero");
            }

            else
                printf("Invalid choice");
        }
    }
}

#include <stdio.h>
void main()
{
    int n, rem, rev = 0, temp;

    printf("Enter a number: ");
    scanf("%d", &n);

    temp = n;

    while (n > 0)
    {
        rem = n % 10;
        rev = rev * 10 + rem;
        n = n / 10;
    }

    if (temp == rev)
        printf("Palindrome");
    else
        printf("Not Palindrome");
}
Output :

Enter a number: 121
Palindrome

#include <stdio.h>
void main()
{
    int n, rem, sum = 0, temp;

    printf("Enter a number: ");
    scanf("%d", &n);

    temp = n;

    while (n > 0)
    {
        rem = n % 10;
        sum = sum + rem * rem * rem;
        n = n / 10;
    }

    if (temp == sum)
        printf("Armstrong Number");
    else
        printf("Not an Armstrong Number");
}
Output :

Enter a number: 153
Armstrong Number
