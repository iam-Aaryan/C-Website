#include <stdio.h>
union Student
{
    int roll_no;
    char name[20];
    int age;
    char class_name[10];
};

void main()
{
    union Student s;

    printf("Enter Roll Number: ");
    scanf("%d", &s.roll_no);
    printf("Roll Number: %d\n", s.roll_no);

    printf("Enter Name: ");
    scanf("%s", s.name);
    printf("Name: %s\n", s.name);

    printf("Enter Age: ");
    scanf("%d", &s.age);
    printf("Age: %d\n", s.age);

    printf("Enter Class: ");
    scanf("%s", s.class_name);
    printf("Class: %s\n", s.class_name);
}
