    gets(str);
    for(i = 0; str[i] != '\0'; i++)
    {
        if(str[i] == 'a' || str[i] == 'e' ||
           str[i] == 'i' || str[i] == 'o' ||

#include <stdio.h>
#include <string.h>
void palindrome()
{
    char str[100], rev[100];
    int i, len;
    printf("Enter a string: ");
    scanf("%s", str);
    len = strlen(str);
    for(i = 0; i < len; i++)
    {
        rev[i] = str[len-i-1];
    }
    rev[i] = '\0';
    if(strcmp(str, rev) == 0)
        printf("String is Palindrome");
    else
        printf("String is not Palindrome");
}
int main()
{
    palindrome();
    return 0;
}
Output :

Enter a string: madam
String is Palindrome
