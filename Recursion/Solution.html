Answer 1


#include <stdio.h>
void display(int n)
{
    if (n > 0)
    {
        printf("%d ", n);
        display(n - 1);
    }
}
void main()
{
    int x=5;
    display(x);
}
Output :

5 4 3 2 1

Answer 2


#include <stdio.h>
int n = 1;
void display()
{
    if (n <= 5)
    {
        printf("%d ", n);
        n++;
        display();
    }
}
void main()
{
    display();
}
Output :

1 2 3 4 5
