#include <stdio.h>
void main()
{
    int arr[5], i, j, temp;
    for(i = 0; i < 5; i++)
    {
        printf("Enter element %d: ", i+1);
        scanf("%d", &arr[i]);
    }
    for(i = 0; i < 5; i++)
    {
        for(j = i+1; j < 5; j++)
        {
            if(arr[i] > arr[j])
            {
                temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
        }
    }
    printf("\nArray in ascending order:\n");
    for(i = 0; i < 5; i++)
    {
        printf("%d\n", arr[i]);
    }
}
Output :

Enter element 1: 50
Enter element 2: 20
Enter element 3: 40
Enter element 4: 10
Enter element 5: 30

Array in ascending order:
10
20
30
40
50
